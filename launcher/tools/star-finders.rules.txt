    match /siteStarFinders/{findingId} {
      allow get: if launcherAdmin() || (request.auth != null && findingId.matches('^' + request.auth.uid + '_[a-z0-9_]{3,16}$'));
      allow list: if launcherAdmin();
      allow update, delete: if false;
      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['uid','nickname','contact','foundAt'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.nickname is string
        && request.resource.data.nickname.matches('^[A-Za-z0-9_]{3,16}$')
        && findingId == request.auth.uid + '_' + request.resource.data.nickname.lower()
        && request.resource.data.contact is string && request.resource.data.contact.size() <= 80
        && request.resource.data.foundAt == request.time;
    }
